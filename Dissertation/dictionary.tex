\chapter*{Словарь терминов}             % Заголовок
\addcontentsline{toc}{chapter}{Словарь терминов}  % Добавляем его в оглавление

$R_i$ : $i$-е автономное роботизированное наземное транспортно-технологическое средство

$\text{Eqp}_i$ : оборудование $R_i$, сокращенно от <<Equipment>>

$E_i$ : ресурс $R_i$, сокращенно от <<Energy>>

$ \mathcal{T}(t) $ : общий групповой массив задач

$\text{Req}_{T_j}$ : соответствие задачи ${T_j}$ типу оборудования $\text{Res}_i$ $R_i$

$\Pi_s(t)$ : внешний поток задач с интенсивностью $\lambda_s$

$\Pi_l(t)$ : внутренний поток задач с интенсивностью $\lambda_l$

$ \Theta_T $ : сложность выполнения задачи $ T $

$ \text{Req}_T $ : требуемое для выполнения задачи $ T $ оборудование, от <<Requirements>>

$ B^{(i)} $ : локальный массив задач для каждого $R_i$ из общего группового массива задач

$ A_i $ : решение, принимаемое $ R_i $

$ \Upsilon^{(i)} $ : действие $ R_i $

$ \mathcal{G} $ : связанный граф взаимодействия $ R $

$ \mathcal{R}_i $ : другие $ R $ в окружении $R_i$ такие, что $ R \in \mathcal{G} $

$ L_{ij} $ : пороговое значение, которое определяет условия равновесия в игре между $R_i$ и $R_j$

$ \mathcal{S}_{ij} $ : множество чистых стратегий равновесия Нэша для $R_i$ и $R_j$

$ \mathcal{U}(a_i, a_j, \theta) $ : полезность $R_i$ в игре с $R_j$

$ \mathcal{V}_i(a_i, a_j, \theta) $ : локальная функция полезности

$ \gamma_i $ : стратегия действия $R_i$

$ \Gamma $ : профиль (набор) стратегии координации

$ \Gamma_{-i} $ : профиль (набор) стратегии координации всех $R$, кроме $R_i$

$ \max_{\gamma_i}\mathcal{J}_i(\gamma_i, \Gamma_{-i}) $ : задача выбора оптимальной стратегии для $R_i$, при которой его ожидаемый выигрыш максимален, при условии, что стратегии других агентов фиксированы
