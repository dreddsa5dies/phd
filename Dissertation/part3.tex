И, учитывая итеративный характер процесса, сформулируем алгоритм для энергосберегающей стратегии:

\begin{breakablealgorithm}
	\caption{Энергосберегающая стратегия}
	\begin{algorithmic}[1]
		\State Инициализировать:
		\State \quad $E_i^{(0)} \sim U[0, E_{\text{max}}]$ \Comment{начальная энергия}
		\State \quad $\Theta_j \sim \mathcal{R}^+(0,\sigma^2)$ \Comment{сложность задач}
		\While{$\exists \text{ невыполненные задачи } T_j$}
		\For{каждого НТТС $R_i \in [N]$}
		\State выбрать $T_j^* = \arg\min\limits_{\{T_j \mid E_i^{(t)} \geq \Theta_j\}} \Theta_j$  \Comment{требуемая задача}
		\State обновить энергию: $E_i^{(t+1)} = E_i^{(t)} - \dfrac{\Theta_j^*}{|\mathcal{C}(T_j^*)|}$
		\EndFor
		\If $\sum\limits_{R_i \in \mathcal{C}(T_j)} \dfrac{\Theta_j}{|\mathcal{C}(T_j)|} \geq \Theta_j$ \Comment{завершенные задачи}
		\State удалить задачу $T_j \in \mathcal{T}$
		\EndIf
		\State $t \leftarrow t + 1$
		\State обновить $\mathcal{T}$
		\EndWhile
	\end{algorithmic}
\end{breakablealgorithm}

\begin{breakablealgorithm}
	\caption{стратегии минимизации задержек}
	\begin{algorithmic}[1]
		\State Инициализировать $W_j^{(0)} = 0$, $\Theta_j \sim \mathcal{R}^+(0,\sigma^2)$
		\For{$t = 1,2,\dotsc$}
		\For{каждого $R_i \in [N]$}
		\State Выбрать
		$T_j^* = \arg\max\limits_{\substack{\{T_j \mid E_i \geq \Theta_j\}}} W_j^{(t)}$
		\State Обновить $W_j^{(t+1)} = 0$ для выбранных задач
		\EndFor
		\State Обновить $W_j^{(t+1)} = W_j^{(t)} + \Delta t$ для не выбранных
		\EndFor
	\end{algorithmic}
\end{breakablealgorithm}

\begin{breakablealgorithm}
	\caption{стратегии баланса ресурсов}
	\begin{algorithmic}[1]
		\State Инициализировать $\mathcal{P}_j \sim U[0,1]$, $\Theta_j \sim \mathcal{R}^+(0,\sigma^2)$
		\For{$t = 1,2,\dotsc$}
		\For{каждого $R_i \in [N]$}
		\State Вычислить
		$\eta_j^{(i)} = \dfrac{\mathcal{P}_j}{\Theta_j} \cdot \dfrac{E_{\text{max}}}{E_i^{(t)}}$
		\State Выбрать
		$T_j^* = \arg\max\limits_{\substack{\{T_j \mid E_i \geq \Theta_j\}}} \eta_j^{(i)}$
		\State Обновить
		$E_i^{(t+1)} = E_i^{(t)} - \dfrac{\Theta_j^*}{1 + |\mathcal{C}(T_j^*)|}$
		\EndFor
		\State Обновить
		$\mathcal{P}_j = \mathcal{P}_j \cdot \left(1 + \dfrac{W_j}{\sum_m W_m}\right)$
		\EndFor
	\end{algorithmic}
\end{breakablealgorithm}
