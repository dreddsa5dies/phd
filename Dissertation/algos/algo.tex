\section{Алгоритм энергосберегающей стратегии}\label{app:A1}

\begin{breakablealgorithm}
	\caption{Энергосберегающая стратегия $\Gamma_{\min\text{Э}}$}
	\begin{algorithmic}[1]
		\State \textbf{Инициализация:}
		\State \quad $E_i^{(0)} \gets E_i^{\text{нач}}$ \Comment{начальная энергия машин}
		\State \quad $\Theta_j \gets \Theta_j^{\text{нач}}$ \Comment{энергетическая стоимость задач}
		\State \quad $\text{Eqp}_i \gets \text{Eqp}_i^{\text{нач}}$ \Comment{оборудование машин}
		\State \quad $\text{Req}_j \gets \text{Req}_j^{\text{нач}}$ \Comment{требуемое оборудование задач}
		\State \quad $\mathcal{E}_{\text{исп}} \gets 0$ \Comment{использованная энергия}
		\State \quad $N_{\text{вып}} \gets 0$ \Comment{количество выполненных задач}
		\State \quad $\text{progress} \gets \text{true}$

		\While{$\text{progress}$}
		\State \quad $\text{progress} \gets \text{false}$
		\ForAll{машина $R_i \in [N]$}
		\If{$E_i > 0$}
		\State \quad \text{Найти задачу $T_j$ с минимальным } $\dfrac{\Theta_j}{E_i}$
		\State \quad среди доступных: $\Theta_j > 0$ \textbf{и} $\text{Eqp}_i \cap \text{Req}_j \neq \emptyset$
		\If{такая задача $T_j$ найдена}
		\State \quad $\delta \gets \min(E_i, \Theta_j)$ \Comment{вклад энергии}
		\State \quad $E_i \gets E_i - \delta$
		\State \quad $\Theta_j \gets \Theta_j - \delta$
		\State \quad $\mathcal{E}_{\text{исп}} \gets \mathcal{E}_{\text{исп}} + \delta$
		\State \quad $\text{progress} \gets \text{true}$
		\If{$\Theta_j \leq 0$}
		\State \quad $\Theta_j \gets 0$
		\State \quad $N_{\text{вып}} \gets N_{\text{вып}} + 1$
		\EndIf
		\EndIf
		\EndIf
		\EndFor
		\EndWhile

		\State \textbf{Вывод результатов:}
		\State \quad $N_{\text{невып}} \gets |\{T_j : \Theta_j > 0\}|$
		\State \quad Вернуть $\mathcal{E}_{\text{исп}}, N_{\text{вып}}, N_{\text{невып}}$
	\end{algorithmic}
\end{breakablealgorithm}

\section{Алгоритм стратегии минимизации временных задержек}\label{app:A2}

\begin{breakablealgorithm}
	\caption{Стратегия минимизации временных задержек $\Gamma_{\min\text{З}}$}
	\begin{algorithmic}[1]
		\State \textbf{Инициализация:}
		\State \quad $E_i^{(0)} \gets E_i^{\text{нач}}$ \Comment{начальная энергия машин}
		\State \quad $\Theta_j \gets \Theta_j^{\text{нач}}$ \Comment{энергетическая стоимость задач}
		\State \quad $\text{Eqp}_i \gets \text{Eqp}_i^{\text{нач}}$ \Comment{оборудование машин}
		\State \quad $\text{Req}_j \gets \text{Req}_j^{\text{нач}}$ \Comment{требуемое оборудование задач}
		\State \quad $W_j \gets \text{возраст задачи } T_j$ \Comment{$W_j = t - t_j^{\text{созд}}$}
		\State \quad $\mathcal{E}_{\text{исп}} \gets 0$ \Comment{использованная энергия}
		\State \quad $N_{\text{вып}} \gets 0$ \Comment{количество выполненных задач}
		\State \quad $P_j \gets 0$ для всех $j$ \Comment{частота выбора задачи на прошлой итерации}
		\State \quad $\alpha, \beta$ \Comment{весовые коэффициенты стратегии}

		\While{остались невыполненные задачи \textbf{и} доступная энергия}
		\State \quad $\text{progress} \gets \text{false}$
		\State \quad $\text{choices}[1..N] \gets -1$ \Comment{выбор задачи каждой машиной}
		\State \quad $\text{currChoices} \gets \{0\}$ \Comment{счётчик выборов задач на этой итерации}

		\ForAll{машина $R_i \in [N]$}
		\If{$E_i > 0$}
		\State \quad Найти задачу $T_j$, минимизирующую:
		\State \quad \quad $V_j = \beta \cdot W_j \cdot \Theta_j - \alpha \cdot P_j$
		\State \quad при условиях: $\Theta_j > 0$ \textbf{и} $\text{Eqp}_i \cap \text{Req}_j \neq \emptyset$
		\If{такая задача найдена}
		\State \quad $\text{choices}[i] \gets j$
		\State \quad $\text{currChoices}[T_j] \gets \text{currChoices}[T_j] + 1$
		\EndIf
		\EndIf
		\EndFor

		\State \quad \textbf{Агрегация выборов:} для каждой задачи $T_j$ — список машин, выбравших её
		\State \quad $\mathcal{C}(T_j) \gets \{ R_i \mid \text{choices}[i] = j \}$

		\State \quad \textbf{Фаза исполнения:}
		\ForAll{задача $T_j$ с $\mathcal{C}(T_j) \neq \emptyset$}
		\State \quad $\text{вклад} \gets 0$
		\ForAll{машина $R_i \in \mathcal{C}(T_j)$}
		\If{$E_i > 0$ \textbf{и} $\text{вклад} < \Theta_j$}
		\State \quad $\delta \gets \min(E_i, \Theta_j - \text{вклад})$
		\State \quad $E_i \gets E_i - \delta$
		\State \quad $\text{вклад} \gets \text{вклад} + \delta$
		\State \quad $\mathcal{E}_{\text{исп}} \gets \mathcal{E}_{\text{исп}} + \delta$
		\State \quad $\text{progress} \gets \text{true}$
		\EndIf
		\EndFor
		\State \quad $\Theta_j \gets \Theta_j - \text{вклад}$
		\If{$\Theta_j \leq 0$}
		\State \quad $\Theta_j \gets 0$
		\State \quad $N_{\text{вып}} \gets N_{\text{вып}} + 1$
		\EndIf
		\EndFor

		\State \quad Обновить $P_j \gets \dfrac{\text{currChoices}[T_j]}{N}$ для всех $j$ \Comment{частота выбора}
		\State \quad Обновить $W_j$ для всех активных задач \Comment{$W_j = \text{текущий возраст}$}

		\If{$\text{not progress}$}
		\State \quad \textbf{break}
		\EndIf
		\EndWhile

		\State \textbf{Вывод результатов:}
		\State \quad $N_{\text{невып}} \gets |\{T_j : \Theta_j > 0\}|$
		\State \quad Вернуть $\mathcal{E}_{\text{исп}}, N_{\text{вып}}, N_{\text{невып}}$
	\end{algorithmic}
\end{breakablealgorithm}

\section{Алгоритм стратегии прямого перебора}\label{app:A3}

\begin{breakablealgorithm}
	\caption{Стратегия прямого перебора $\Gamma_{\text{прям}}$}
	\begin{algorithmic}[1]
		\State \textbf{Инициализация:}
		\State \quad $E_i^{(0)} \gets E_i^{\text{нач}}$ \Comment{начальная энергия машин}
		\State \quad $\Theta_j \gets \Theta_j^{\text{нач}}$ \Comment{энергетическая стоимость задач}
		\State \quad $\text{Eqp}_i \gets \text{Eqp}_i^{\text{нач}}$ \Comment{оборудование машин}
		\State \quad $\text{Req}_j \gets \text{Req}_j^{\text{нач}}$ \Comment{требуемое оборудование задач}
		\State \quad $\mathcal{E}_{\text{исп}} \gets 0$ \Comment{использованная энергия}
		\State \quad $N_{\text{вып}} \gets 0$ \Comment{количество выполненных задач}
		\State \quad $\text{progress} \gets \text{true}$

		\While{$\text{progress}$}
		\State \quad $\text{progress} \gets \text{false}$
		\ForAll{машина $R_i \in [N]$}
		\If{$E_i > 0$}
		\ForAll{задача $T_j \in [M]$}
		\If{$\Theta_j > 0$ \textbf{and} $\text{Eqp}_i \cap \text{Req}_j \neq \emptyset$}
		\State \quad $\delta \gets \min(E_i, \Theta_j)$ \Comment{размер вклада}
		\State \quad $\Theta_j \gets \Theta_j - \delta$
		\State \quad $E_i \gets E_i - \delta$
		\State \quad $\mathcal{E}_{\text{исп}} \gets \mathcal{E}_{\text{исп}} + \delta$
		\State \quad $\text{progress} \gets \text{true}$
		\If{$\Theta_j \leq 0$}
		\State \quad $N_{\text{вып}} \gets N_{\text{вып}} + 1$
		\EndIf
		\State \quad \textbf{break} \Comment{машина работает только с одной задачей за итерацию}
		\EndIf
		\EndFor
		\EndIf
		\EndFor
		\EndWhile

		\State \textbf{Вывод результатов:}
		\State \quad $N_{\text{невып}} \gets |\{T_j : \Theta_j > 0\}|$
		\State \quad Вернуть $\mathcal{E}_{\text{исп}}, N_{\text{вып}}, N_{\text{невып}}$
	\end{algorithmic}
\end{breakablealgorithm}

\section{Алгоритм стратегии RL (Q-learning)}\label{app:A4}

\begin{breakablealgorithm}
	\caption{Стратегия Q-обучения $\Gamma_{\text{QL}}$}
	\begin{algorithmic}[1]
		\State \textbf{Инициализация:}
		\State \quad $Q \gets \emptyset$ \Comment{Q-таблица: $Q[m_i][a] = q$}
		\State \quad $\alpha \gets \text{LR}$ \Comment{коэффициент обучения}
		\State \quad $\gamma \gets \text{Gamma}$ \Comment{коэффициент дисконтирования}
		\State \quad $\varepsilon \gets \text{Epsilon}$ \Comment{начальное значение $\varepsilon$-жадности}
		\State \quad $\varepsilon_{\min} \gets \text{MinEps}$ \Comment{минимальное $\varepsilon$}
		\State \quad $\delta \gets \text{Decay}$ \Comment{коэффициент затухания $\varepsilon$}
		\State \quad $\mathcal{E}_{\text{исп}} \gets 0$ \Comment{использованная энергия}
		\State \quad $N_{\text{вып}} \gets 0$ \Comment{количество выполненных задач}
		\State \quad $\text{doneSet} \gets \emptyset$ \Comment{множество завершённых задач}

		\State \textbf{Подготовка:}
		\State \quad $\mathcal{E}_{\text{нач}} \gets \sum_{i} E_i$ \Comment{общая начальная энергия}
		\State \quad $\text{maxIters} \gets 10000 + 10 \cdot M \cdot \max(1, N)$ \Comment{макс. итераций}

		\For{$\text{iter} = 1$ \textbf{to} $\text{maxIters}$}
		\State \quad \textbf{if} $\sum E_i = 0$ \textbf{or} $\sum \Theta_j = 0$ \textbf{then break} \Comment{стоп-критерии}
		\State \quad $\text{progress} \gets \text{false}$

		\ForAll{машина $R_i \in [N]$}
		\If{$E_i > 0$}
		\State \quad \text{Доступные действия: } $A_i \gets \emptyset$
		\State \quad $\text{taskIdxByAction} \gets \emptyset$
		\ForAll{задача $T_j \in [M]$}
		\If{$\Theta_j > 0$ \textbf{and} $\text{Eqp}_i \cap \text{Req}_j \neq \emptyset$}
		\State \quad $a \gets \text{Req}_j$ \Comment{тип требуемого оборудования}
		\State \quad $A_i \gets A_i \cup \{a\}$
		\State \quad $\text{taskIdxByAction}[a] \gets \text{taskIdxByAction}[a] \cup \{j\}$
		\EndIf
		\EndFor

		\If{$A_i = \emptyset$}
		\State \quad \textbf{continue}
		\EndIf

		\State \quad \textbf{Инициализация $Q[R_i]$ при необходимости}
		\State \quad \textbf{Выбор действия $a^*$ по $\varepsilon$-жадной политике:}
		\If{$\text{rand}() < \varepsilon$}
		\State \quad $a^* \gets \text{случайное из } A_i$
		\Else
		\State \quad $a^* \gets \arg\max_{a \in A_i} Q[R_i][a]$
		\EndIf

		\State \quad \text{Выбор задачи $T_j^*$ из $\text{taskIdxByAction}[a^*]$ с минимальным $\Theta_j$}
		\State \quad $\delta \gets \min(E_i, \Theta_j^*)$ \Comment{вклад энергии}
		\State \quad $E_i \gets E_i - \delta$
		\State \quad $\Theta_j^* \gets \Theta_j^* - \delta$
		\State \quad $\mathcal{E}_{\text{исп}} \gets \mathcal{E}_{\text{исп}} + \delta$
		\State \quad $\text{progress} \gets \text{true}$

		\State \quad \textbf{Расчёт награды $r$:}
		\If{$\Theta_j^* \leq 0$}
		\State \quad $r \gets 10.0 + 0.01 \cdot \delta$ \Comment{премия за завершение}
		\State \quad $N_{\text{вып}} \gets N_{\text{вып}} + 1$
		\State \quad $\text{doneSet} \gets \text{doneSet} \cup \{T_j^*\}$
		\Else
		\State \quad $r \gets 0.01 \cdot \delta - 0.001 \cdot \delta$ \Comment{награда за прогресс минус штраф}
		\EndIf

		\State \quad \textbf{Обновление Q-значения:}
		\State \quad $Q_{\text{old}} \gets Q[R_i][a^*]$
		\State \quad $Q_{\text{max}}' \gets \max_{a \in A_i} Q[R_i][a]$ \Comment{макс. Q для следующего шага}
		\State \quad $Q[R_i][a^*] \gets Q_{\text{old}} + \alpha \cdot (r + \gamma \cdot Q_{\text{max}}' - Q_{\text{old}})$
		\EndIf
		\EndFor

		\State \quad \textbf{Обновление $\varepsilon$:}
		\State \quad \textbf{if} $\varepsilon > \varepsilon_{\min}$ \textbf{then} $\varepsilon \gets \max(\varepsilon \cdot \delta, \varepsilon_{\min})$

		\State \quad \textbf{if not progress} \textbf{then break}
		\EndFor

		\State \textbf{Вывод результатов:}
		\State \quad $\mathcal{E}_{\text{исп}} \gets \mathcal{E}_{\text{нач}} - \sum_{i} E_i$
		\State \quad $N_{\text{невып}} \gets |\{T_j : \Theta_j > 0\}|$
		\State \quad Вернуть $\mathcal{E}_{\text{исп}}, N_{\text{вып}}, N_{\text{невып}}$
	\end{algorithmic}
\end{breakablealgorithm}

\section{Алгоритм стратегии рыночного распределения}\label{app:A5}

\begin{breakablealgorithm}
	\caption{Стратегия рыночного распределения $\Gamma_{\text{рын}}$}
	\begin{algorithmic}[1]
		\State \textbf{Инициализация:}
		\State \quad $\mathcal{E}_{\text{нач}} \gets \sum_{i=1}^{N} E_i$ \Comment{начальная суммарная энергия}
		\State \quad $\mathcal{E}_{\text{исп}} \gets 0$ \Comment{израсходованная энергия}
		\State \quad $N_{\text{вып}} \gets 0$ \Comment{количество завершённых задач}
		\State \quad $\text{doneSet} \gets \emptyset$ \Comment{множество выполненных задач}

		\ForAll{задача $T_j \in [M]$}
		\If{$\Theta_j > 0$}
		\State \quad \text{Сбор ставок от машин:}
		\State \quad $B_j \gets \emptyset$ \Comment{множество ставок для $T_j$}
		\ForAll{машина $R_i \in [N]$}
		\If{$E_i > 0$ \textbf{and} $\text{Eqp}_i \cap \text{Req}_j \neq \emptyset$}
		\State \quad $c_{ij} \gets \dfrac{\Theta_j}{E_i + \varepsilon}$ \Comment{стоимость ставки, $\varepsilon = 10^{-6}$}
		\State \quad $B_j \gets B_j \cup \{(i, c_{ij})\}$
		\EndIf
		\EndFor

		\If{$B_j = \emptyset$}
		\State \quad \textbf{continue} \Comment{некому выполнять задачу}
		\EndIf

		\State \quad Отсортировать $B_j$ по возрастанию $c_{ij}$ \Comment{лучшие ставки — в начале}
		\State \quad $(i^*, c^*) \gets B_j[1]$ \Comment{лучшая машина по ставке}

		\If{$E_{i^*} \geq \Theta_j$}
		\State \quad $\delta \gets \Theta_j$ \Comment{машина полностью закрывает задачу}
		\State \quad $E_{i^*} \gets E_{i^*} - \delta$
		\State \quad $\Theta_j \gets 0$
		\State \quad $N_{\text{вып}} \gets N_{\text{вып}} + 1$
		\State \quad $\text{doneSet} \gets \text{doneSet} \cup \{T_j\}$
		\Else
		\State \quad \text{Распределение $T_j$ между несколькими машинами:}
		\State \quad $L \gets$ список машин из $B_j$ в порядке возрастания $c_{ij}$
		\State \quad $\text{оставшееся} \gets \Theta_j$
		\State \quad $A \gets \sum_{i \in L} E_i$ \Comment{суммарная доступная энергия}
		\If{$A > 0$}
		\ForAll{машина $R_i \in L$ \textbf{и} $\text{оставшееся} > 0$}
		\If{$E_i > 0$}
		\State \quad $\text{доля} \gets (E_i / A) \cdot \Theta_j$
		\State \quad $\delta \gets \min(\text{доля}, E_i, \text{оставшееся})$
		\State \quad $E_i \gets E_i - \delta$
		\State \quad $\text{оставшееся} \gets \text{оставшееся} - \delta$
		\EndIf
		\EndFor
		\State \quad $\Theta_j \gets \text{оставшееся}$
		\If{$\Theta_j \leq 0$}
		\State \quad $\Theta_j \gets 0$
		\State \quad $N_{\text{вып}} \gets N_{\text{вып}} + 1$
		\State \quad $\text{doneSet} \gets \text{doneSet} \cup \{T_j\}$
		\EndIf
		\EndIf
		\EndIf
		\EndIf
		\EndFor

		\State \textbf{Вывод результатов:}
		\State \quad $\mathcal{E}_{\text{исп}} \gets \mathcal{E}_{\text{нач}} - \sum_{i=1}^{N} E_i$
		\State \quad $N_{\text{невып}} \gets |\{T_j : \Theta_j > 0\}|$
		\State \quad Вернуть $\mathcal{E}_{\text{исп}}, N_{\text{вып}}, N_{\text{невып}}$
	\end{algorithmic}
\end{breakablealgorithm}
