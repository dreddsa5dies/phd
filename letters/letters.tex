\def\setfonts#1{%
  \mscale=#1\relax
  \font\myfont={LiberationSerif-Regular.ttf} at \dimexpr 14pt * \mscale / 1000 \relax
  \font\myfontit={LiberationSerif-Italic.ttf} at \dimexpr 14pt * \mscale / 1000 \relax
  \myfont
}

\def\rebuildboxes{%
  \setbox\tobox=\vbox{%
    \hsize=\tohsize
    \parindent=0pt
    \raggedleft
    {\myfontit Кому:}
    \ifx\towhom\empty\else\towhom\break\fi
    \toinstitution \par
    {\myfontit Куда:}
    \toaddress \break
    \tocity \break
    \toindex
  }%
  \needed=\dimexpr
    \ht\tobox  + \dp\tobox
    + \idxreserve
    + 10mm \relax
}

\setbox\idxbox=\hbox{\postindex\$\toindex}%
\ifdim\wd\idxbox>\fullhsize
  \setbox\idxbox=\hbox{\resizebox{\fullhsize}{!}{\postindex\$\toindex}}%
\fi
\idxreserve=\dimexpr \ht\idxbox + \dp\idxbox + 8mm \relax

\setfonts{1000}%
\rebuildboxes
\loop
  \ifdim\needed>\vsize
    \ifnum\mscale>800
      \advance\mscale by -50
      \setfonts{\mscale}%
      \rebuildboxes
    \else
      \mscale=1
    \fi
  \else
    \mscale=1
  \fi
\ifnum\mscale>1 \repeat
