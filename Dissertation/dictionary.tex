\chapter*{Словарь терминов}             % Заголовок
\addcontentsline{toc}{chapter}{Словарь терминов}  % Добавляем его в оглавление

$R_i$ : $i$-е автономное роботизированное наземное транспортно-технологическое средство

$\text{Eqp}_i$ : тип оборудования $R_i$, от англ.\ <<Equipment>>

$\text{Eqp}_i \succeq \text{Req}_{T_j}$ : оборудование $R_i$ удовлетворяет или превосходит требования задачи $T_j$

$E_i$ : доступный объем энергетического ресурса $R_i$, от англ.\ <<Energy>>

$\mathcal{T}(t)$ : групповой массив задач в момент времени $t$: $\mathcal{T}(t) = \{ T_1, \dotsc, T_{M(t)} \}$

$\text{Req}_{T_j}$ : требуемый тип оборудования для выполнения задачи $T_j$, от англ.\ <<Requirements>>

$\Pi_s(t)$ : внешний пуассоновский поток задач с интенсивностью $\lambda_s$

$\Pi_l(t)$ : внутренний пуассоновский поток задач с интенсивностью $\lambda_l$

$\Theta_T$ : сложность выполнения задачи $T$, выраженная в количестве необходимого ресурса

$\text{Req}_T$ : то же, что $\text{Req}_{T_j}$, используется как краткая форма

$\mathcal{N}_{[0, +\infty)}(\mu_T, \sigma_T^2)$ : усечённое нормальное распределение на положительной полуоси, из которого выбирается $\Theta_T$

$B^{(i)}$ : бинарный вектор локального выбора задач для $R_i$: $b_j^{(i)} = 1$ если $T_j$ доступна для $R_i$

$A_i$ : действие $R_i$, соответствует выбору конкретной задачи: $A_i = T_j$

$\mathbf{1}(A_i = T_j)$ : Индикаторная функция: 1, если $R_i$ выбрал задачу $T_j$, и 0 в противном случае

$\mathbf{A} = (A_1,\dots,A_N)$ : вектор действий всех НТТС в системе

$\Upsilon^{(i)}$ : наблюдаемый сигнал о возможных действиях, используемый $R_i$ для принятия решения

$\mathcal{G}$ : связанный неориентированный граф взаимодействия НТТС: $\mathcal{G} = ([N], \mathcal{E})$

$\mathcal{E}$ : множество ребер графа $\mathcal{G}$, определяющее связи между НТТС

$\mathcal{R}_i$ : множество соседей $R_i$ в графе $\mathcal{G}$

$\mathcal{S}_{ij}$ : множество чистых стратегий равновесия Нэша для пары $R_i$ и $R_j$

$\kappa_{ij}$ : степень кооперации между $R_i$ и $R_j$, определяемая как $\frac{|\mathcal{R}_i \cap \mathcal{R}_j|}{|\mathcal{R}_i| + |\mathcal{R}_j|}$

$\alpha$ : выигрыш за успешную координацию усилий двух НТТС

$\beta$ : штраф за сложность выполнения задачи

$\mathcal{U}(A_i, A_j, T)$ : функция полезности пары $R_i$ и $R_j$ при совместном выборе задач

$\mathcal{V}_i(\mathbf{A}, \Theta)$ : локальная функция полезности $R_i$, зависящая от действий соседей и сложностей задач

$\mathbf{E}_{\Theta, \mathbf{A}_{-i}}[\cdot]$ : математическое ожидание по случайным параметрам: сложностям задач $\Theta$ и действиям других НТТС $\mathbf{A}_{-i}$

$\gamma_i$ : стратегия $R_i$, отображающая наблюдение $\Upsilon^{(i)}$ в действие $A_i$

$\Gamma$ : профиль стратегии координации всех НТТС ($\Gamma = (\gamma_1, \dotsc, \gamma_N)$)

$\Gamma_{-i}$ : профиль стратегии координации всех НТТС, кроме $R_i$

$\max_{\gamma_i}\mathcal{J}_i(\gamma_i, \Gamma_{-i})$ : задача максимизации ожидаемой полезности $R_i$ при фиксированных стратегиях других НТТС

$\Gamma_{min\text{Э}}$ : профиль стратегии, минимизирующий энергетические затраты

$\Gamma_{min\text{ВЗ}}$ : профиль стратегии, минимизирующий время выполнения задач

$V_i(\mathbf{A})$ : матрица игры для $\Gamma$

$W_j(t)$ : время ожидание задачи $j$

$\underline{v}$ : нижняя цена игры -- максимальная гарантированная выигрышная величина для $R_i$, если он предполагает наихудший случай, то есть когда $R_j$ старается минимизировать его выигрыш (лучший из худших исходов для $R_i$)

$\overline{v}$ : верхняя цена игры -- минимальная потеря, которую может себе гарантировать $R_j$, предполагая, что $R_i$ будет стремиться максимизировать свой выигрыш (худший из лучших исходов для $R_j$)

$v_s$ : цена игры при симметрии, нижняя и верхняя цены игры совпадают (оптимальная $\Gamma$)

$\mathbb{E}[W_k]$ : ожидаемое (среднее) время ожидания задачи $T_k$, усреднённое по всем возможным случайным состояниям системы

$\mathbb{P}\left( \bigcap_{j \in \mathcal{R}_i} \{ A_j = k \} \right)$ : вероятности того, что все соседи $j$ из окружения $R_i$ ​выберут одну и ту же задачу $k$

$\mathbb{E}[\Theta]$ : математическое ожидание (среднее значение) сложности задачи в системе

$W_{\text{eq}}$ : время ожидания в состоянии равновесия
